## target/27b/meas.sh
Script (verbatim), used for the before/B/C/D measurement passes on git/git (85,649 commits):

    #!/bin/bash
    # usage: meas.sh <label>  — 3 runs each nocg and cg; appends summary lines
    cd <home____________>/dev/sandbox/gitter
    L=$1; CG=$(scripts/clone-git-git.sh --where); NOCG=target/27b/nocg.git
    for which in nocg cg; do
      R=$NOCG; [ $which = cg ] && R=$CG
      for i in 1 2 3; do
        out=$( { /usr/bin/time -l target/release/examples/budgets "$R"; } 2>&1 )
        fp=$(echo "$out" | awk '/peak memory footprint/ {printf "%.1f", $1/1048576}')
        line=$(echo "$out" | grep '^first paint' | sed -E 's/.*first paint ([^,]*), G ([^,]*),.*end ([0-9.]+) MiB/paint \1 G \2 rss \3/')
        echo "$L $which run$i: $line footprint $fp" | tee -a target/27b/meas/summary.txt
      done
    done
